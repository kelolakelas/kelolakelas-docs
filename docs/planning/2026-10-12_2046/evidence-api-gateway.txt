28|// ClientIPTrust decides where the gateway reads the client IP from (KEL-62).
29|//
30|// The client IP keys the rate limiter and is recorded by the access log, so both
31|// always read it from the same place: gin's ClientIP, configured here once.
32|//
33|// With the zero value no proxy is trusted and the client IP is the socket
34|// address of the peer, whatever headers the request carries. With TrustedProxies
35|// and Header set, a request whose socket peer lies inside TrustedProxies may name
36|// the client in Header. The header is read right to left and the first address
37|// that is not itself a trusted proxy wins, so a caller cannot prepend a forged
38|// address in front of a trusted chain. An empty header, a value that is not an
39|// address (including an IPv6 address with a zone), or a peer outside
40|// TrustedProxies falls back to the socket address.
41|type ClientIPTrust struct {
42|	TrustedProxies []string
43|	Header         string
44|}

298|// applyClientIPTrust configures gin's ClientIP from the trust policy. The zero
299|// policy keeps SetTrustedProxies(nil) with no forwarded header, so every request
300|// is identified by its socket address exactly as before KEL-62.
301|func applyClientIPTrust(r *gin.Engine, trust ClientIPTrust) error {
302|	// Platform headers (Cloudflare, App Engine, …) are never trusted implicitly;
303|	// they would bypass the proxy check entirely.
304|	r.TrustedPlatform = ""
305|	if len(trust.TrustedProxies) == 0 || trust.Header == "" {
306|		r.ForwardedByClientIP = false
307|		r.RemoteIPHeaders = nil
308|		return r.SetTrustedProxies(nil)
309|	}
310|	if err := r.SetTrustedProxies(trust.TrustedProxies); err != nil {
311|		return fmt.Errorf("trusted proxies: %w", err)
312|	}
313|	r.ForwardedByClientIP = true
314|	r.RemoteIPHeaders = []string{trust.Header}
315|	return nil
