299|  const match = generatorTimePattern.exec(value.trim());

307|  const hours = Math.floor(totalMinutes / 60);

312|/**
313| * Generates weekly recurring slots from day, time-window, session-length, and
314| * break choices (KEL-112).
315| *
316| * Sessions run back-to-back from the window start with `session_minutes`
317| * duration separated by `break_minutes`. A session that would end after the
318| * window end is skipped, so a session ending exactly on the boundary is kept.
319| * Days are de-duplicated and sorted, and anything invalid (malformed time,
320| * non-positive session, negative break, an end that is not after the start,
321| * no usable day) yields no slots instead of throwing — the form validates
322| * first and treats an empty result as "nothing fits". An overnight window is
323| * unsupported and also yields no slots. Capacity and the trimmed location are
324| * carried onto every produced slot unchanged.
325| */
326|export function generateWeeklySlots(
327|  input: WeeklySlotGeneratorInput
328|): GeneratedWeeklySlot[] {
329|  const start = parseWallClockMinutes(input.start_time);
330|  const end = parseWallClockMinutes(input.end_time);
331|
332|  if (start === null || end === null || end <= start) {
333|    return [];
334|  }
335|
336|  if (
337|    !Number.isInteger(input.session_minutes) ||
338|    input.session_minutes <= 0 ||
339|    !Number.isInteger(input.break_minutes) ||
340|    input.break_minutes < 0
341|  ) {
342|    return [];
343|  }
344|
345|  const days = [...new Set(input.days)]
346|    .filter((day) => Number.isInteger(day) && day >= 1 && day <= 7)
347|    .sort((a, b) => a - b);
348|
349|  if (days.length === 0) {
350|    return [];
351|  }
352|
353|  const location = input.location?.trim() ? input.location.trim() : undefined;
354|  const slots: GeneratedWeeklySlot[] = [];
355|
356|  for (const day of days) {
357|    let cursor = start;
358|    while (cursor + input.session_minutes <= end) {
359|      slots.push({
360|        day_of_week: day,
361|        start_time: formatWallClockMinutes(cursor),
362|        end_time: formatWallClockMinutes(cursor + input.session_minutes),
363|        capacity: input.capacity,
364|        ...(location ? { location } : {}),
365|      });
366|      cursor += input.session_minutes + input.break_minutes;
367|    }
368|  }
369|
370|  return slots;

406|/**
407| * Finds every pair of slots that overlap on the same day (KEL-176).
408| *
409| * Two slots overlap when they share `day_of_week` and their half-open
410| * intervals intersect (`[start, end)`): a slot ending exactly when another
411| * starts only touches and is not an overlap. Pairs with an unparseable or
412| * half-typed time (an empty or malformed bound) and intervals whose end is
413| * not after their start are skipped instead of throwing, because the form
414| * re-runs this on every keystroke while the tenant is still typing.
415| */
416|export function findOverlappingSlotPairs(
417|  slots: readonly SlotTimeRange[]
418|): OverlappingSlotPair[] {
419|  const pairs: OverlappingSlotPair[] = [];
420|
421|  for (let first = 0; first < slots.length; first += 1) {
422|    for (let second = first + 1; second < slots.length; second += 1) {
423|      const a = slots[first];
424|      const b = slots[second];
425|
426|      if (a.day_of_week !== b.day_of_week) {
427|        continue;
428|      }
429|
430|      const aStart = parseSlotTimeSeconds(a.start_time);
431|      const aEnd = parseSlotTimeSeconds(a.end_time);
432|      const bStart = parseSlotTimeSeconds(b.start_time);
433|      const bEnd = parseSlotTimeSeconds(b.end_time);
434|
435|      if (
436|        aStart === null ||
437|        aEnd === null ||
438|        bStart === null ||
439|        bEnd === null ||
440|        aEnd <= aStart ||
441|        bEnd <= bStart
442|      ) {
443|        continue;
444|      }
445|
446|      if (aStart < bEnd && bStart < aEnd) {
447|        pairs.push({ first, second });
448|      }
449|    }
450|  }
451|
452|  return pairs;

4|type Caller = { tenant_id?: unknown; is_parent?: unknown; is_platform_admin?: unknown; platform_factor_version?: unknown };

6|export function creatorCaller(token: string): Caller | null {
7|  try {
8|    const encoded = token.split('.')[1];
9|    if (!encoded) return null;
10|    const normalized = encoded.replace(/-/g, '+').replace(/_/g, '/');
11|    const payload = JSON.parse(atob(normalized.padEnd(Math.ceil(normalized.length / 4) * 4, '=')));
12|    return typeof payload === 'object' && payload !== null ? payload as Caller : null;
13|  } catch { return null; }
14|}

16|export function callerMatchesScope(token: string, platform: boolean, tenant?: string): boolean {
17|  const caller = creatorCaller(token);
18|  if (!caller) return false;
19|  if (platform) return caller.is_platform_admin === true && !caller.tenant_id && typeof caller.platform_factor_version === 'number' && caller.platform_factor_version > 0;
20|  return caller.is_platform_admin !== true && caller.is_parent !== true && typeof tenant === 'string' && tenant.length > 0 && caller.tenant_id === tenant;
