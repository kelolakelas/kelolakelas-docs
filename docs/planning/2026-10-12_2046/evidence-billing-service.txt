53|// NormalizeVoucherCode canonicalizes a voucher code for storage and
54|// comparison. Codes are unique case-insensitively per tenant (KEL-161), and
55|// the database constraint is an exact match, so new code writes store the
56|// canonical form: trimmed and uppercased. Existing mixed-case rows remain
57|// valid; the write path checks for them under a tenant-scoped transaction lock.
58|func NormalizeVoucherCode(code string) string {
59|	return strings.ToUpper(strings.TrimSpace(code))
60|}

62|// ValidateVoucherFields checks the shared create/update semantics: a
63|// percentage is 1-100, a nominal is positive, the date range is ordered, and
64|// the usage cap never drops below what was already consumed.
65|func ValidateVoucherFields(code, discountType string, discountValue float64, minTransactionAmount int64, maxDiscountAmount *int64, maxUses *int, currentUses int, validFrom, validUntil *time.Time) error {
66|	if strings.TrimSpace(code) == "" || len([]rune(code)) > 255 {
67|		return ErrVoucherInvalid
68|	}
69|	switch discountType {
70|	case VoucherDiscountPercentage:
71|		if discountValue < 1 || discountValue > 100 {
72|			return ErrVoucherInvalid
73|		}
74|	case VoucherDiscountFixedAmount:
75|		if discountValue <= 0 {
76|			return ErrVoucherInvalid
77|		}
78|	default:
79|		return ErrVoucherInvalid
80|	}
81|	if minTransactionAmount < 0 {
82|		return ErrVoucherInvalid
83|	}
84|	if maxDiscountAmount != nil && *maxDiscountAmount <= 0 {
85|		return ErrVoucherInvalid
86|	}
87|	if currentUses < 0 {
88|		return ErrVoucherInvalid
89|	}
90|	if maxUses != nil && (*maxUses <= 0 || *maxUses < currentUses) {
91|		return ErrVoucherInvalid
92|	}
93|	if validFrom != nil && validUntil != nil && validUntil.Before(*validFrom) {
94|		return ErrVoucherInvalid
95|	}
96|	return nil
97|}

66|func (u *voucherUsecase) Create(ctx context.Context, tenantID uuid.UUID, req *domain.CreateVoucherRequest) (*domain.VoucherResponse, error) {
67|	if req == nil {
68|		return nil, domain.ErrVoucherInvalid
69|	}
70|	code := domain.NormalizeVoucherCode(req.Code)
71|	if err := domain.ValidateVoucherFields(code, req.DiscountType, req.DiscountValue, req.MinTransactionAmount, req.MaxDiscountAmount, req.MaxUses, 0, req.ValidFrom, req.ValidUntil); err != nil {
72|		return nil, err
73|	}
74|	if req.MinTransactionAmount < 0 {
75|		return nil, domain.ErrVoucherInvalid
76|	}
77|	isActive := true
78|	if req.IsActive != nil {
79|		isActive = *req.IsActive
80|	}
81|	voucher := &domain.Voucher{
82|		ID:                   uuid.New(),
83|		TenantID:             tenantID,
84|		Code:                 code,
85|		DiscountType:         req.DiscountType,
86|		DiscountValue:        req.DiscountValue,
87|		MaxDiscountAmount:    req.MaxDiscountAmount,
88|		MinTransactionAmount: req.MinTransactionAmount,
89|		MaxUses:              req.MaxUses,
90|		IsActive:             isActive,
91|		ValidFrom:            utcTimestamp(req.ValidFrom),
92|		ValidUntil:           utcTimestamp(req.ValidUntil),
93|	}
94|	create := func(ctx context.Context) error {
95|		if err := u.vouchers.LockTenantCodes(ctx, tenantID); err != nil {
96|			return err
97|		}
98|		// Existing rows may predate canonical storage. Check under the tenant
99|		// lock so concurrent code writes cannot both miss a legacy variant.
100|		if err := u.requireCodeFree(ctx, tenantID, uuid.Nil, code); err != nil {
101|			return err
102|		}
103|		return u.vouchers.Create(ctx, voucher)
104|	}
105|	if u.txManager != nil {
106|		if err := u.txManager.WithTransaction(ctx, create); err != nil {
107|			return nil, mapVoucherWriteError(err)
108|		}
109|	} else if err := create(ctx); err != nil {
110|		return nil, mapVoucherWriteError(err)
111|	}
112|	return domain.ToVoucherResponse(voucher), nil

123|func (u *voucherUsecase) Update(ctx context.Context, tenantID, id uuid.UUID, req *domain.UpdateVoucherRequest) (*domain.VoucherResponse, error) {
124|	if id == uuid.Nil || req == nil {
125|		return nil, domain.ErrVoucherInvalid
126|	}
127|	if req.Code == nil && req.DiscountType == nil && req.DiscountValue == nil && req.MaxDiscountAmount == nil &&
128|		req.MinTransactionAmount == nil && req.MaxUses == nil && req.ValidFrom == nil && req.ValidUntil == nil && req.IsActive == nil {
129|		return nil, domain.ErrVoucherInvalid
130|	}
131|	var voucher *domain.Voucher
132|	update := func(ctx context.Context) error {
133|		// Acquire the tenant code lock before the row lock, in the same order
134|		// as other code writes, to avoid deadlocks during concurrent renames.
135|		if req.Code != nil {
136|			if err := u.vouchers.LockTenantCodes(ctx, tenantID); err != nil {
137|				return err
138|			}
139|		}
140|		var err error
141|		voucher, err = u.lockedVoucher(ctx, tenantID, id)
142|		if err != nil {
143|			return err
144|		}
145|		code := voucher.Code
146|		if req.Code != nil {
147|			code = domain.NormalizeVoucherCode(*req.Code)
148|		}
149|		discountType := voucher.DiscountType
150|		if req.DiscountType != nil {
151|			discountType = *req.DiscountType
152|		}
153|		discountValue := voucher.DiscountValue
154|		if req.DiscountValue != nil {
155|			discountValue = *req.DiscountValue
156|		}
157|		maxDiscount := voucher.MaxDiscountAmount
158|		if req.MaxDiscountAmount != nil {
159|			maxDiscount = req.MaxDiscountAmount
160|		}
161|		minAmount := voucher.MinTransactionAmount
162|		if req.MinTransactionAmount != nil {
163|			minAmount = *req.MinTransactionAmount
164|		}
165|		maxUses := voucher.MaxUses
166|		if req.MaxUses != nil {
167|			maxUses = req.MaxUses
168|		}
169|		validFrom := voucher.ValidFrom
170|		if req.ValidFrom != nil {
171|			validFrom = utcTimestamp(req.ValidFrom)
172|		}
173|		validUntil := voucher.ValidUntil
174|		if req.ValidUntil != nil {
175|			validUntil = utcTimestamp(req.ValidUntil)
176|		}
177|		// The usage cap is validated against the locked current_uses, so a
178|		// concurrent checkout redemption (KEL-29) that consumed the last use
179|		// cannot slip under a lowered cap between the read and the write.
180|		if err := domain.ValidateVoucherFields(code, discountType, discountValue, minAmount, maxDiscount, maxUses, voucher.CurrentUses, validFrom, validUntil); err != nil {
181|			return err
182|		}
183|		if req.Code != nil {
184|			if err := u.requireCodeFree(ctx, tenantID, id, code); err != nil {
185|				return err
186|			}
187|			voucher.Code = code
188|		}
189|		voucher.DiscountType = discountType
190|		voucher.DiscountValue = discountValue
191|		voucher.MaxDiscountAmount = maxDiscount
192|		voucher.MinTransactionAmount = minAmount
193|		voucher.MaxUses = maxUses
194|		voucher.ValidFrom = validFrom
195|		voucher.ValidUntil = validUntil
196|		if req.IsActive != nil {
197|			voucher.IsActive = *req.IsActive
198|		}
199|		voucher.UpdatedAt = time.Now().UTC()
200|		return u.vouchers.Update(ctx, voucher)
201|	}
202|	if u.txManager != nil {
203|		if err := u.txManager.WithTransaction(ctx, update); err != nil {
204|			return nil, mapVoucherWriteError(err)
205|		}
206|	} else if err := update(ctx); err != nil {
207|		return nil, mapVoucherWriteError(err)
208|	}
209|	return domain.ToVoucherResponse(voucher), nil

212|// Delete removes a voucher that never discounted a transaction. A voucher
213|// with CurrentUses > 0 answers ErrVoucherInUse: history must survive, so it
214|// can only be deactivated through Update, never deleted.
215|func (u *voucherUsecase) Delete(ctx context.Context, tenantID, id uuid.UUID) error {
216|	if id == uuid.Nil {
217|		return domain.ErrVoucherInvalid
218|	}
219|	var inUse bool
220|	remove := func(ctx context.Context) error {
221|		voucher, err := u.lockedVoucher(ctx, tenantID, id)
222|		if err != nil {
223|			return err
224|		}
225|		if voucher.CurrentUses > 0 {
226|			inUse = true
227|			return domain.ErrVoucherInUse
228|		}
229|		deleted, err := u.vouchers.SoftDelete(ctx, tenantID, id)
230|		if err != nil {
231|			return err
232|		}
233|		if !deleted {
234|			return domain.ErrVoucherNotFound
235|		}
236|		return nil
237|	}
238|	if u.txManager != nil {
239|		if err := u.txManager.WithTransaction(ctx, remove); err != nil {
240|			if errors.Is(err, domain.ErrVoucherInUse) {
241|				return domain.ErrVoucherInUse
242|			}
243|			return mapVoucherWriteError(err)
244|		}
245|		return nil
246|	}
247|	if err := remove(ctx); err != nil {
248|		if errors.Is(err, domain.ErrVoucherInUse) {
249|			return domain.ErrVoucherInUse
250|		}
251|		return mapVoucherWriteError(err)
252|	}
253|	if inUse {
254|		return domain.ErrVoucherInUse
255|	}
256|	return nil

298|// mapVoucherWriteError translates storage rejections: the unique backstop
299|// firing on a concurrent create with the same code is a duplicate, whatever
300|// case the two writers used.
301|func mapVoucherWriteError(err error) error {
302|	if err == nil {
303|		return nil
304|	}
305|	if errors.Is(err, domain.ErrVoucherNotFound) || errors.Is(err, domain.ErrVoucherInvalid) ||
306|		errors.Is(err, domain.ErrVoucherDuplicate) || errors.Is(err, domain.ErrVoucherInUse) {
307|		return err
308|	}
309|	if errors.Is(err, gorm.ErrRecordNotFound) {
310|		return domain.ErrVoucherNotFound
311|	}
312|	msg := err.Error()
313|	if containsConstraint(msg, "uq_vouchers_tenant_code") || containsConstraint(msg, "duplicate key value violates unique constraint") {
314|		return domain.ErrVoucherDuplicate
315|	}
316|	return err
317|}
318|
319|func containsConstraint(msg, sub string) bool {
320|	if len(msg) < len(sub) {
321|		return false
322|	}
323|	for i := 0; i+len(sub) <= len(msg); i++ {
324|		if msg[i:i+len(sub)] == sub {
325|			return true
326|		}
327|	}
328|	return false
